fun BusinessCard(
    name: String,
    title: String,
    phone: String,
    email: String,
    github: String
) {
    printDivider()
    LogoSection()
    HeaderSection(name, title)
    ContactSection(phone, email, github)
    printDivider()
}

fun LogoSection() {
    println("        ◼◼◼◼◼◼◼")
    println("        ◼  KB  ◼")
    println("        ◼◼◼◼◼◼◼")
    println()
}

fun HeaderSection(name: String, title: String) {
    println("        $name")
    println("        $title")
    println()
}

fun ContactSection(phone: String, email: String, github: String) {
    ContactItem("📞", phone)
    ContactItem("✉️", email)
    ContactItem("🌐", github)
}

fun ContactItem(icon: String, value: String) {
    println("   $icon  $value")
}

fun printDivider() {
    println("=======================================")
}
